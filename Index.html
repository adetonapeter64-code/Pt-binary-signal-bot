// XAUUSD trend-pullback signal bot -> Telegram (H4 candles) // Run: TG_TOKEN=xxx TG_CHAT_ID=yyy node index.js // Data: Yahoo gold futures (GC=F) as a proxy for spot XAUUSD. import fs from "node:fs"; const TOKEN = process.env.TG_TOKEN; const CHAT_ID = process.env.TG_CHAT_ID; if (!TOKEN || !CHAT_ID) { console.error("Set TG_TOKEN and TG_CHAT_ID environment variables."); process.exit(1); } const SYMBOL = "GC=F"; const SL_ATR = 1.5; const RR = 2; const CHECK_EVERY_MS = 15 * 60 * 1000; const STATE_FILE = "last_signal.txt"; async function getH4() { const url = `https://query1.finance.yahoo.com/v8/finance/chart/${encodeURIComponent( SYMBOL )}?interval=1h&range=60d`; const res = await fetch(url, { headers: { "User-Agent": "Mozilla/5.0" } }); const json = await res.json(); const r = json.chart.result[0]; const q = r.indicators.quote[0]; const buckets = new Map(); r.timestamp.forEach((t, i) => { if (q.close[i] == null) return; const key = Math.floor(t / 14400) * 14400; // 4-hour buckets (UTC) const b = buckets.get(key); if (!b) { buckets.set(key, { t: key, high: q.high[i], low: q.low[i], close: q.close[i] }); } else { b.high = Math.max(b.high, q.high[i]); b.low = Math.min(b.low, q.low[i]); b.close = q.close[i]; } }); return [...buckets.values()]; } function ema(values, span) { const a = 2 / (span + 1); const out = [values[0]]; for (let i = 1; i < values.length; i++) out.push(a * values[i] + (1 - a) * out[i - 1]); return out; } function wilder(values, period) { const a = 1 / period; const out = [values[0]]; for (let i = 1; i < values.length; i++) out.push(a * values[i] + (1 - a) * out[i - 1]); return out; } function indicators(c) { const close = c.map((x) => x.close); const gains = [0], losses = [0], tr = [c[0].high - c[0].low]; for (let i = 1; i < c.length; i++) { const d = close[i] - close[i - 1]; gains.push(Math.max(d, 0)); losses.push(Math.max(-d, 0)); tr.push( Math.max( c[i].high - c[i].low, Math.abs(c[i].high - close[i - 1]), Math.abs(c[i].low - close[i - 1]) ) ); } const ag = wilder(gains, 14); const al = wilder(losses, 14); return { ema50: ema(close, 50), ema200: ema(close, 200), rsi: ag.map((g, i) => 100 - 100 / (1 + g / al[i])), atr: wilder(tr, 14), }; } function checkSignal(c) { const n = c.length; if (n < 210) return null; const ind = indicators(c); const i = n - 2; // last CLOSED candle (the final one is still forming) const p = n - 3; const recent = c.slice(n - 7, n - 1); const close = c[i].close; const up = close > ind.ema200[i] && ind.ema50[i] > ind.ema200[i]; const down = close < ind.ema200[i] && ind.ema50[i] < ind.ema200[i]; const risk = SL_ATR * ind.atr[i]; if (up && Math.min(...recent.map((x) => x.low)) <= ind.ema50[i] && ind.rsi[p] < 45 && ind.rsi[i] >= 45) return { t: c[i].t, side: "BUY", entry: close, sl: close - risk, tp: close + RR * risk }; if (down && Math.max(...recent.map((x) => x.high)) >= ind.ema50[i] && ind.rsi[p] > 55 && ind.rsi[i] <= 55) return { t: c[i].t, side: "SELL", entry: close, sl: close + risk, tp: close - RR * risk }; return null; } async function send(text) { await fetch(`https://api.telegram.org/bot${TOKEN}/sendMessage`, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ chat_id: CHAT_ID, text }), }); } async function tick() { try { const sig = checkSignal(await getH4()); const last = fs.existsSync(STATE_FILE) ? fs.readFileSync(STATE_FILE, "utf8") : ""; if (sig && String(sig.t) !== last) { await send( `XAUUSD ${sig.side} (H4)\nEntry: ${sig.entry.toFixed(2)}\nSL: ${sig.sl.toFixed( 2 )}\nTP: ${sig.tp.toFixed(2)}\nRisk max 1% of account.` ); fs.writeFileSync(STATE_FILE, String(sig.t)); } } catch (e) { console.error("Error:", e.message); } } await send("XAUUSD pullback bot started (H4)."); await tick(); setInterval(tick, CHECK_EVERY_MS);
